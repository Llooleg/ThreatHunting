``` r
    tryCatch({
      sample_val <- unnested_data[[col]][[which(!sapply(unnested_data[[col]], is.null))[1]]]
      
      if (is.list(sample_val) && !is.null(names(sample_val))) {
        unnested_data <- unnested_data %>% 
          unnest_wider(all_of(col), names_sep = "_", simplify = TRUE, strict = FALSE)
      } else {
        unnested_data[[col]] <- sapply(unnested_data[[col]], function(x) {
          if (is.null(x) || length(x) == 0) return(NA_character_)
          if (length(x) == 1) return(as.character(x))
          paste(x, collapse = ", ")
        })
      }
    }, error = function(e) {
      cat("    Ошибка в колонке", col, "- пропускаем\n")
    })
  }
  
  list_cols <- names(unnested_data)[sapply(unnested_data, is.list)]
}
```

``` r
detect_and_convert <- function(col) {
  if (is.numeric(col) || is.logical(col) || inherits(col, "POSIXct")) {
    return(col)
  }
  
  if (!is.character(col)) return(col)
  
  non_na <- na.omit(col)
  if (length(non_na) == 0) return(col)
  
  sample_size <- min(1000, length(non_na))
  sample_vals <- non_na[1:sample_size]
  
  if (all(tolower(sample_vals) %in% c("true", "false", "1", "0"))) {
    return(case_when(
      tolower(col) %in% c("true", "1") ~ TRUE,
      tolower(col) %in% c("false", "0") ~ FALSE,
      TRUE ~ NA
    ))
  }
  
  if (all(grepl("^-?[0-9]+$", sample_vals))) {
    return(as.integer(col))
  }
  
  if (all(grepl("^-?[0-9]+(\\.[0-9]+)?$", sample_vals))) {
    return(as.numeric(col))
  }
  
  if (any(grepl("^\\d{4}-\\d{2}-\\d{2}", sample_vals[1:min(10, length(sample_vals))]))) {
    parsed <- lubridate::parse_date_time(col, 
                                          orders = c("ymd HMS", "ymd HM", "dmy HMS"),
                                          quiet = TRUE)
    if (sum(!is.na(parsed)) / length(col) > 0.5) {
      return(parsed)
    }
  }
  
  return(col)
}
```

    Сколько их?

``` r
event_col_candidates <- grep("event.*code|event.*id|^event_id$", names(minimized_data), 
                             ignore.case = TRUE, value = TRUE)

if (length(event_col_candidates) > 0) {
  event_col_name <- event_col_candidates[1]
  cat("Используем колонку с кодами событий:", event_col_name, "\n")
  
  # Убеждаемся что numeric
  if (!is.numeric(minimized_data[[event_col_name]])) {
    minimized_data[[event_col_name]] <- as.numeric(minimized_data[[event_col_name]])
  }
  
  # Подсчёт с агрегацией
  event_summary <- minimized_data %>%
    count(.data[[event_col_name]], name = "count") %>%
    rename(event_code = 1) %>%
    left_join(event_df, by = c("event_code" = "event_id"))
  
  criticality_counts <- event_summary %>%
    summarise(
      high = sum(count[grepl("High", criticality, ignore.case = TRUE)], na.rm = TRUE),
      medium = sum(count[grepl("Medium", criticality, ignore.case = TRUE)], na.rm = TRUE),
      total_events = sum(count)
    )
  
  cat("События высокой значимости (High):   ", criticality_counts$high, "\n")
  cat("События средней значимости (Medium): ", criticality_counts$medium, "\n")
  cat("Всего событий в датасете:            ", criticality_counts$total_events, "\n")
}
```
